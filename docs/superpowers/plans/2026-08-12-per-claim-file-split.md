**Goal:** Split `837 Sample/fromUHCG-837I-WKN_9claims.x12` into nine standalone X12 interchanges, one claim each, with claim data copied byte-for-byte and the nine-claim file left untouched.

**Architecture:** A new `tests/split_claims.py` reads the nine-claim file read-only, copies each `ST..SE` transaction set verbatim, and wraps it in an envelope cloned from the source `ISA`/`GS` with a per-file interchange control number (`ISA13 = 78010023N`) and group control number (`GS06 = N`). A new `tests/split_roundtrip.py` proves byte fidelity and envelope correctness; the existing 24-check validator gains `--claim-index` and `--group-control` so it can grade a single-claim file against claim N's expectations; a new `tests/split_mutations.py` proves those checks actually catch tampering.

**Tech Stack:** Python 3.11 standard library only (`argparse`, `io`, `os`, `hashlib`, `re`, `shutil`, `subprocess`, `tempfile`, `decimal`). Shared X12 reader: `tests/x12_parse.py`. No third-party packages, no test framework — every test is a runnable script that exits 0 on success and 1 on failure, matching the existing suite.

"""Prove the per-claim files carry the nine-claim file's data byte-for-byte."""
from __future__ import annotations
import argparse
import hashlib
import io
import os
import sys

import x12_parse as X

HERE = os.path.dirname(os.path.abspath(__file__))
DEFAULT_INPUT = os.path.join(HERE, "..", "837 Sample",
                             "fromUHCG-837I-WKN_9claims.x12")
DEFAULT_OUTDIR = os.path.join(HERE, "..", "837 Sample", "WKN_9claims_split")
NAME = "fromUHCG-837I-WKN_claim%02d.x12"
ENVELOPE = ("ISA*", "GS*", "GE*", "IEA*")

results = []


def check(num, name, ok, detail=""):
    results.append((num, name, bool(ok), detail))


def sha256(path):
    return hashlib.sha256(io.open(path, "rb").read()).hexdigest()


def block_raw(raw):
    """The ST..SE slice of a file's raw text, terminator included."""
    start = raw.index("ST*")
    end = raw.index("~", raw.index("SE*", start)) + 1
    return raw[start:end]


def run(args):
    in_raw, in_segs = X.read_segments(args.input)
    in_sets = X.transaction_sets(in_segs)
    before = sha256(args.input)

    paths = [os.path.join(args.outdir, NAME % n) for n in range(1, 10)]
    missing = [p for p in paths if not os.path.exists(p)]
    check(1, "nine per-claim files exist", not missing,
          "missing: %s" % [os.path.basename(p) for p in missing])
    if missing:
        for num in range(2, 7):
            check(num, "not evaluated (files missing)", False, "see check 1")
        check(7, "input sha256 unchanged", sha256(args.input) == before, before)
        return

    outs = [X.read_segments(p) for p in paths]

    # 2. one ST..SE per file, correctly nested
    nest_ok, nest_detail = True, ""
    for n, (_raw, segs) in enumerate(outs, start=1):
        sets = X.transaction_sets(segs)
        if len(sets) != 1:
            nest_ok = False
            nest_detail += "file %d has %d ST..SE sets; " % (n, len(sets))
    check(2, "exactly one ST..SE transaction set per file", nest_ok, nest_detail)

    # 3. claim payload identical, segment for segment, to the matching input set
    seg_ok, seg_detail = True, ""
    for n, (_raw, segs) in enumerate(outs, start=1):
        st, se = X.transaction_sets(segs)[0]
        want = in_segs[in_sets[n - 1][0]:in_sets[n - 1][1] + 1]
        got = segs[st:se + 1]
        if got != want:
            seg_ok = False
            diff = [(a, b) for a, b in zip(want, got) if a != b][:1]
            seg_detail += "file %d differs (%d vs %d segments) %r; " % (
                n, len(want), len(got), diff)
    check(3, "each file's ST..SE matches the input transaction set segment for segment",
          seg_ok, seg_detail)

    # 4. the same thing at the byte level: the raw ST..SE text occurs in the input
    raw_ok, raw_detail = True, ""
    for n, (raw, _segs) in enumerate(outs, start=1):
        if block_raw(raw) not in in_raw:
            raw_ok = False
            raw_detail += "file %d ST..SE bytes not found in the input; " % n
    check(4, "each file's ST..SE bytes appear verbatim in the nine-claim file",
          raw_ok, raw_detail)

    # 5. no claim segment lost or duplicated across the nine files
    in_body = [s for s in in_segs if not s.startswith(ENVELOPE)]
    out_body = []
    for _raw, segs in outs:
        out_body.extend(s for s in segs if not s.startswith(ENVELOPE))
    check(5, "the nine files together hold every non-envelope segment, once",
          in_body == out_body,
          "input %d segments, split %d segments" % (len(in_body), len(out_body)))

    # 6. envelope controls
    env_ok, env_detail = True, ""
    icns = []
    for n, (_raw, segs) in enumerate(outs, start=1):
        e = {s.split("*")[0]: X.elems(s) for s in segs
             if s.startswith(ENVELOPE)}
        # A per-file flag: a missing segment in file 3 must not skip file 4.
        complete = True
        for sid in ("ISA", "GS", "GE", "IEA"):
            if sid not in e:
                env_ok = complete = False
                env_detail += "file %d missing %s; " % (n, sid)
        if not complete:
            continue
        isa13, gs06, ge01, ge02, iea01, iea02 = (
            e["ISA"][13], e["GS"][6], e["GE"][1], e["GE"][2],
            e["IEA"][1], e["IEA"][2])
        icns.append(isa13)
        if len(isa13) != 9:
            env_ok = False
            env_detail += "file %d ISA13 %r is not 9 characters; " % (n, isa13)
        if iea02 != isa13:
            env_ok = False
            env_detail += "file %d IEA02 %s != ISA13 %s; " % (n, iea02, isa13)
        if gs06 != str(n):
            env_ok = False
            env_detail += "file %d GS06 %s != %d; " % (n, gs06, n)
        if ge02 != gs06:
            env_ok = False
            env_detail += "file %d GE02 %s != GS06 %s; " % (n, ge02, gs06)
        if ge01 != "1" or iea01 != "1":
            env_ok = False
            env_detail += "file %d GE01=%s IEA01=%s, both must be 1; " % (
                n, ge01, iea01)
    if len(set(icns)) != len(icns):
        env_ok = False
        env_detail += "interchange control numbers repeat: %s; " % icns
    check(6, "unique ISA13 per file, IEA02=ISA13, GS06=N, GE02=GS06, GE01=IEA01=1",
          env_ok, env_detail)

    # 7. the input was not touched
    check(7, "input sha256 unchanged by the split", sha256(args.input) == before,
          before)


def report():
    width = max(len(n) for _num, n, _ok, _d in results)
    failed = 0
    for num, name, ok, detail in results:
        if ok:
            print("[PASS] %2d. %-*s" % (num, width, name))
        else:
            failed += 1
            print("[FAIL] %2d. %-*s  %s" % (num, width, name, detail))
    print("\n%d checks, %d passed, %d failed"
          % (len(results), len(results) - failed, failed))
    return 1 if failed else 0


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--input", default=DEFAULT_INPUT)
    ap.add_argument("--outdir", default=DEFAULT_OUTDIR)
    args = ap.parse_args()
    run(args)
    return report()


if __name__ == "__main__":
    sys.exit(main())

```python
"""Split the nine-claim UHCG 837I test file into nine standalone interchanges.

The ST..SE payload of each claim is copied verbatim; only the envelope differs,
so each file carries a unique interchange and group control number the way a
production submission would.

Usage:
    python split_claims.py [--input PATH] [--outdir PATH] [--force]
"""
from __future__ import annotations
import argparse
import io
import os
import sys

import x12_parse as X

HERE = os.path.dirname(os.path.abspath(__file__))
DEFAULT_INPUT = os.path.join(HERE, "..", "837 Sample",
                             "fromUHCG-837I-WKN_9claims.x12")
DEFAULT_OUTDIR = os.path.join(HERE, "..", "837 Sample", "WKN_9claims_split")
NAME = "fromUHCG-837I-WKN_claim%02d.x12"
ISA13_PREFIX = "78010022"
EXPECTED_SETS = 9


def set_elem(seg, idx, val):
    e = X.elems(seg)
    while len(e) <= idx:
        e.append("")
    e[idx] = val
    return "*".join(e)


def build_files(segs):
    """Return [(claim_index, [segment, ...]), ...], one standalone interchange each."""
    if not segs or not segs[0].startswith("ISA*"):
        raise ValueError("first segment is not ISA")
    if len(segs) < 2 or not segs[1].startswith("GS*"):
        raise ValueError("second segment is not GS")
    sets = X.transaction_sets(segs)
    if len(sets) != EXPECTED_SETS:
        raise ValueError("expected %d ST..SE sets, found %d"
                         % (EXPECTED_SETS, len(sets)))
    built = []
    for n, (st, se) in enumerate(sets, start=1):
        icn = "%s%d" % (ISA13_PREFIX, n)
        if len(icn) != 9:
            raise ValueError("ISA13 %r is not 9 characters" % icn)
        isa = set_elem(segs[0], 13, icn)
        gs = set_elem(segs[1], 6, str(n))
        payload = segs[st:se + 1]  # verbatim, SE01 included and still correct
        built.append((n, [isa, gs] + payload + ["GE*1*%d" % n, "IEA*1*%s" % icn]))
    return built


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--input", default=DEFAULT_INPUT)
    ap.add_argument("--outdir", default=DEFAULT_OUTDIR)
    ap.add_argument("--force", action="store_true")
    args = ap.parse_args()

    src = os.path.abspath(args.input)
    paths = [os.path.join(os.path.abspath(args.outdir), NAME % n)
             for n in range(1, 10)]
    if src in paths:
        sys.exit("refusing to write over the input file")
    existing = [p for p in paths if os.path.exists(p)]
    if existing and not args.force:
        sys.exit("output already exists; pass --force to overwrite: %s"
                 % ", ".join(os.path.basename(p) for p in existing))

    _raw, segs = X.read_segments(src)
    try:
        built = build_files(segs)
    except ValueError as exc:
        sys.exit("refusing to split: %s" % exc)

    # Everything is built before anything is written: a rejected input leaves no
    # half-written set of files behind.
    if not os.path.isdir(os.path.dirname(paths[0])):
        os.makedirs(os.path.dirname(paths[0]))
    for (n, out_segs), path in zip(built, paths):
        io.open(path, "w", encoding="utf-8", newline="").write(
            "".join(s + "~\n" for s in out_segs))
        print("wrote %s (%d segments)" % (os.path.basename(path), len(out_segs)))


if __name__ == "__main__":
    main()
```

from __future__ import annotations
import os
import subprocess
import sys

            [sys.executable, os.path.join(HERE, "validate_9claims.py"), path,
             "--expect-claims", "1", "--claim-index", str(n),
             "--group-control", str(n)],

- [ ] **Step 1: Write the failing test**

Create `tests/split_mutations.py`:

```python
"""Mutate throwaway copies of the split output; assert the right check catches each one.

VALIDATOR mutations edit one per-claim file and must produce a specific [FAIL] n
from validate_9claims.py. ROUNDTRIP mutations must make split_roundtrip.py fail:
its checks are what guarantee the split did not lose, reorder or duplicate data.
"""
from __future__ import annotations
import io
import os
import re
import shutil
import subprocess
import sys
import tempfile

HERE = os.path.dirname(os.path.abspath(__file__))
SRC_INPUT = os.path.join(HERE, "..", "837 Sample", "fromUHCG-837I-WKN_9claims.x12")
SRC_OUTDIR = os.path.join(HERE, "..", "837 Sample", "WKN_9claims_split")
NAME = "fromUHCG-837I-WKN_claim%02d.x12"

# (label, claim file to mutate, old, new, expected failing validator check)
VALIDATOR_MUTATIONS = [
    ("dropped_line_date", 3, "DTP*472*D8*20241123~\n", "", 22),
    ("line_hcp_amount_changed", 4,
     "HCP*10*357.00*63.00*9327*0.00**0.00****UN*2~\n",
     "HCP*10*358.00*63.00*9327*0.00**0.00****UN*2~\n", 8),
    ("st02_from_another_claim", 5, "ST*837*010649905*", "ST*837*010649907*", 10),
    ("se01_decremented", 6, "SE*55*010649906~", "SE*54*010649906~", 11),
    ("last_line_block_dropped", 7,
     "LX*7~\nSV2*0440*HC:92507*384.00*UN*1~\nDTP*472*D8*20241127~\n"
     "HCP*10*326.40*57.60*9327*0.00**0.00****UN*1~\n", "", 6),
    ("group_control_wrong", 8, "*8*X*005010X223A2~", "*3*X*005010X223A2~", 21),
]

# (label, claim file to mutate, old, new)
ROUNDTRIP_MUTATIONS = [
    ("duplicate_interchange_control", 2, "*780100222*", "*780100221*"),
    ("payload_segment_deleted", 9, "DTP*096*TM*0000~\n", ""),
    ("iea02_out_of_step", 4, "IEA*1*780100224~", "IEA*1*780100229~"),
]


def stage():
    """Copy the nine split files into a fresh temp directory."""
    tmp = tempfile.mkdtemp(prefix="x12split-")
    for n in range(1, 10):
        shutil.copyfile(os.path.join(SRC_OUTDIR, NAME % n),
                        os.path.join(tmp, NAME % n))
    return tmp


def mutate(tmp, claim, old, new):
    """Apply exactly one substitution; return True if it actually changed the file."""
    path = os.path.join(tmp, NAME % claim)
    raw = io.open(path, encoding="utf-8", newline="").read()
    if raw.count(old) < 1:
        return False
    mutated = raw.replace(old, new, 1)
    if mutated == raw:
        return False
    io.open(path, "w", encoding="utf-8", newline="").write(mutated)
    return True


def run_validator(tmp, claim):
    r = subprocess.run(
        [sys.executable, os.path.join(HERE, "validate_9claims.py"),
         os.path.join(tmp, NAME % claim),
         "--expect-claims", "1", "--claim-index", str(claim),
         "--group-control", str(claim)],
        capture_output=True, text=True)
    return sorted(int(m) for m in re.findall(r"\[FAIL\]\s+(\d+)\.", r.stdout)), r.returncode


def run_roundtrip(tmp):
    r = subprocess.run(
        [sys.executable, os.path.join(HERE, "split_roundtrip.py"),
         "--input", SRC_INPUT, "--outdir", tmp],
        capture_output=True, text=True)
    return sorted(int(m) for m in re.findall(r"\[FAIL\]\s+(\d+)\.", r.stdout)), r.returncode


def main():
    # A mutation is only evidence if the unmutated copy is clean.
    base = stage()
    for n in range(1, 10):
        fails, rc = run_validator(base, n)
        if fails or rc != 0:
            print("BASELINE NOT CLEAN: claim %d fails %s (exit %d)" % (n, fails, rc))
            return 1
    rt_fails, rt_rc = run_roundtrip(base)
    if rt_fails or rt_rc != 0:
        print("BASELINE NOT CLEAN: round-trip fails %s (exit %d)" % (rt_fails, rt_rc))
        return 1
    print("baseline: nine clean files, round-trip clean\n")

    uncaught, broken = [], []
    print("mutation                        kind        expect  caught     failing checks")
    for label, claim, old, new, expect in VALIDATOR_MUTATIONS:
        tmp = stage()
        if not mutate(tmp, claim, old, new):
            print("%-31s %-11s BROKEN - pattern made no change" % (label, "validator"))
            broken.append(label)
            continue
        fails, _rc = run_validator(tmp, claim)
        ok = expect in fails
        if not ok:
            uncaught.append(label)
        print("%-31s %-11s %-7d %-10s %s"
              % (label, "validator", expect, "YES" if ok else "*** NO ***", fails))

    for label, claim, old, new in ROUNDTRIP_MUTATIONS:
        tmp = stage()
        if not mutate(tmp, claim, old, new):
            print("%-31s %-11s BROKEN - pattern made no change" % (label, "roundtrip"))
            broken.append(label)
            continue
        fails, rc = run_roundtrip(tmp)
        ok = bool(fails) and rc != 0
        if not ok:
            uncaught.append(label)
        print("%-31s %-11s %-7s %-10s %s"
              % (label, "roundtrip", "any", "YES" if ok else "*** NO ***", fails))

    print("\n%d mutations, %d uncaught, %d broken"
          % (len(VALIDATOR_MUTATIONS) + len(ROUNDTRIP_MUTATIONS),
             len(uncaught), len(broken)))
    return 1 if (uncaught or broken) else 0


if __name__ == "__main__":
    sys.exit(main())
```
